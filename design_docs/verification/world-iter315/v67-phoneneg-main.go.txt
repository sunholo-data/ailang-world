// Iteration-315 designer: synthetic phone negatives for the r3 phone rule (doc §4.4 (iii′)).
// usage: phoneneg <phone.png> <outdir>
// Writes: no-scrollbar.png   bottom 20 rows repainted, column by column, with that column's row-823 colour
//         title-unclipped.png the header band rows 45..67, x 372..389 repainted with each column's row-44 colour
//         no-overflow.png    both of the above (what a non-overflowing page would show)
//         crop-370.png       the leftmost 370 columns only (370x844)
//         pad-370.png        x >= 370 repainted with each row's x=369... no: with the colour at (389, 30) (page bg)
//         full-thumb.png     thumb rows 833..840 extended to x 3..387 with the thumb colour at (100,836)
//         shift-{2,4,5,10}.png  V58b mkshift logic (content moved down N px, top filled with the colour at (5,5))
//         timeline-only.png, blank.png  V58 mkwrong logic
package main

import (
	"fmt"
	"image"
	"image/color"
	"image/draw"
	"image/png"
	"os"
	"path/filepath"
)

func save(dir, name string, im image.Image) {
	o, err := os.Create(filepath.Join(dir, name))
	if err != nil {
		panic(err)
	}
	if err := png.Encode(o, im); err != nil {
		panic(err)
	}
	o.Close()
	fmt.Println("wrote", name)
}

func clone(src image.Image) *image.NRGBA {
	w := image.NewNRGBA(src.Bounds())
	draw.Draw(w, src.Bounds(), src, image.Point{}, draw.Src)
	return w
}

func noScrollbar(w *image.NRGBA) {
	b := w.Bounds()
	for x := 0; x < b.Dx(); x++ {
		c := w.At(x, b.Dy()-21)
		for y := b.Dy() - 20; y < b.Dy(); y++ {
			w.Set(x, y, c)
		}
	}
}

func unclip(w *image.NRGBA) {
	for x := 372; x < 390; x++ {
		c := w.At(x, 44)
		for y := 45; y < 68; y++ {
			w.Set(x, y, c)
		}
	}
}

func main() {
	f, _ := os.Open(os.Args[1])
	src, err := png.Decode(f)
	if err != nil {
		panic(err)
	}
	dir := os.Args[2]
	b := src.Bounds()

	w := clone(src)
	noScrollbar(w)
	save(dir, "no-scrollbar.png", w)

	w = clone(src)
	unclip(w)
	save(dir, "title-unclipped.png", w)

	w = clone(src)
	noScrollbar(w)
	unclip(w)
	save(dir, "no-overflow.png", w)

	c := image.NewNRGBA(image.Rect(0, 0, 370, b.Dy()))
	draw.Draw(c, c.Bounds(), src, image.Point{}, draw.Src)
	save(dir, "crop-370.png", c)

	w = clone(src)
	bg := src.At(389, 30)
	draw.Draw(w, image.Rect(370, 0, b.Dx(), b.Dy()), &image.Uniform{bg}, image.Point{}, draw.Src)
	save(dir, "pad-370.png", w)

	w = clone(src)
	tc := src.At(100, 836)
	for y := 833; y <= 840; y++ {
		for x := 3; x <= 387; x++ {
			w.Set(x, y, tc)
		}
	}
	save(dir, "full-thumb.png", w)

	for _, n := range []int{2, 4, 5, 10} {
		s := image.NewNRGBA(b)
		draw.Draw(s, b, &image.Uniform{src.At(5, 5)}, image.Point{}, draw.Src)
		draw.Draw(s, image.Rect(0, n, b.Dx(), b.Dy()), src, image.Pt(0, 0), draw.Src)
		save(dir, fmt.Sprintf("shift-%d.png", n), s)
	}

	t := image.NewNRGBA(b)
	draw.Draw(t, b, &image.Uniform{src.At(5, 330)}, image.Point{}, draw.Src)
	draw.Draw(t, image.Rect(0, 0, b.Dx(), b.Dy()-330), src, image.Pt(0, 330), draw.Src)
	save(dir, "timeline-only.png", t)

	bl := image.NewNRGBA(b)
	draw.Draw(bl, b, &image.Uniform{color.NRGBA{250, 250, 250, 255}}, image.Point{}, draw.Src)
	save(dir, "blank.png", bl)
}
