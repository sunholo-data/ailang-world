import sys
D = sys.argv[1]  # .../proto/host/broker
def edit(fn, pairs):
    p = D + '/' + fn
    s = open(p).read()
    for a, b in pairs:
        assert s.count(a) == 1, (fn, a[:90], s.count(a))
        s = s.replace(a, b)
    open(p, 'w').write(s)
