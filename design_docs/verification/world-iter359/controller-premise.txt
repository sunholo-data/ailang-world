import socket, time, sys, threading
# Bind N idle loopback listeners, count unsolicited accepts for T seconds.
N, T = 8, 75
socks, counts = [], [0]*8
for i in range(N):
    s = socket.socket(); s.bind(("127.0.0.1", 0)); s.listen(16); s.settimeout(0.2); socks.append(s)
end = time.time() + T
while time.time() < end:
    for i, s in enumerate(socks):
        try:
            c, _ = s.accept(); counts[i] += 1; c.close()
        except socket.timeout:
            pass
print(f"listeners={N} seconds={T} unsolicited_accepts_per_listener={counts} total={sum(counts)}")
