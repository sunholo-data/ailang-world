#!/bin/bash
# Controller only, OUTSIDE the planner sandbox. Compatible with Bash 3.2.
# Usage: bash /path/to/plan-mutants.sh.txt TREE_ROOT
# 21 scenarios (CONTROL + all 20 measured r2 mutants), 177 independent rows.
# Run sequentially with no concurrent source edits or test suites.
# Runner-owned writes stay under $1. Existing probes hard-code /tmp/claude;
# literal process-wide write confinement needs OS path isolation (plan finding F1).
set -euo pipefail
if [ "$#" -ne 1 ]; then
  echo 'usage: bash plan-mutants.sh.txt TREE_ROOT' >&2
  exit 2
fi
root=$(cd "$1" && pwd -P)
cd "$root"
src=host/broker/exec_probe.go
[ -f go.mod ] && [ -f "$src" ] && [ -f host/broker/exec_probe_net6_dial_test.go ]
[ ! -L host ] && [ ! -L host/broker ] && [ ! -L "$src" ]
command -v python3 >/dev/null
command -v go >/dev/null
state=$(mktemp -d "$root/.plan-mutants.XXXXXX")
runtime="$state/runtime"
mkdir "$runtime"
cp "$src" "$state/exec_probe.go.pristine"
restore() {
  cp "$state/exec_probe.go.pristine" "$src" || return 1
  cmp "$state/exec_probe.go.pristine" "$src" || return 1
}
finish() {
  saved_rc=$?
  trap - EXIT HUP INT TERM
  if restore; then
    echo 'RESTORE cmp ok: exit cleanup'
  else
    echo 'RESTORE MISMATCH: exit cleanup' >&2
    saved_rc=3
  fi
  # Module/toolchain caches can contain .go sources. Remove ALL of them, even
  # on failure, before any repo-wide source scan or later suite can run.
  chmod -R u+w "$runtime" 2>/dev/null || true
  rm -rf "$runtime" || saved_rc=3
  rm -f "$state/exec_probe.go.pristine" || saved_rc=3
  if ! python3 -I - "$root" "$state/go-files-before.json" <<'CENSUS'
from pathlib import Path
import json, sys
root, saved = Path(sys.argv[1]), Path(sys.argv[2])
before = set(json.loads(saved.read_text()))
after = {str(p.relative_to(root)) for p in root.rglob('*.go') if p.is_file()}
if after != before:
    raise SystemExit('GO FILE CENSUS MISMATCH: added=%r removed=%r' % (sorted(after-before), sorted(before-after)))
print('GO FILE CENSUS unchanged; no extra .go files remain')
CENSUS
  then saved_rc=3; fi
  echo "Mutation evidence: $state (logs retained; pristine and runtime caches removed)"
  exit "$saved_rc"
}
# Save only non-.go artifacts; never duplicate a Go declaration under this tree.

import json, sys
root = Path(sys.argv[1])
Path(sys.argv[2]).write_text(json.dumps(sorted(str(p.relative_to(root)) for p in root.rglob('*.go') if p.is_file()))+'\n')
CENSUS
trap finish EXIT
trap 'exit 130' INT
trap 'exit 143' TERM
trap 'exit 129' HUP

export GOCACHE="$runtime/gocache" GOMODCACHE="$runtime/modcache" GOPATH="$runtime/gopath"
export XDG_CONFIG_HOME="$runtime/config" XDG_CACHE_HOME="$runtime/cache" GOTELEMETRYDIR="$runtime/telemetry"
export GOENV=off GOWORK=off GOFLAGS= GOCACHEPROG=

rows=0
mismatches=0
run_row() { # mutant, exactly one top-level test, expected rc (0 green / 1 red), assertion text
  mut=$1
  name=$2
  expected=$3
  needle=$4
  restore || { echo "RESTORE MISMATCH: $mut $name precondition" >&2; exit 3; }
  mutate "$mut"
  log="$state/$mut.$name.log"
  rc=0
  go test -count=1 -run "^(${name})$" ./host/broker/ -v -timeout 300s > "$log" 2>&1 || rc=$?
  matched=1
  [ "$rc" -eq "$expected" ] || matched=0
  if [ "$expected" -eq 0 ]; then
    /usr/bin/grep -Fq -- "--- PASS: $name " "$log" || matched=0
  else
    /usr/bin/grep -Fq -- "--- FAIL: $name " "$log" || matched=0
    /usr/bin/grep -Fq -- "$needle" "$log" || matched=0
  fi
  # A build error, missing test, SKIP, timeout or crash is never an expected red.
  if /usr/bin/grep -Eq -- '--- SKIP:|\[build failed\]|^panic:|^fatal error:|no tests to run' "$log"; then
    matched=0
  fi
  rows=$((rows+1))
  if [ "$matched" -eq 1 ]; then
    echo "EXPECTATION MATCH: $mut tests=$name expected_rc=$expected actual_rc=$rc"
  else
    mismatches=$((mismatches+1))
    echo "EXPECTATION MISMATCH: $mut tests=$name expected_rc=$expected actual_rc=$rc log=$log"
    cat "$log"
  fi
  restore || { echo "RESTORE MISMATCH: $mut $name" >&2; exit 3; }
  echo "RESTORE cmp ok: $mut tests=$name"
}
