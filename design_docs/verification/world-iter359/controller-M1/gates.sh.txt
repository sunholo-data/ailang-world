#!/bin/bash
# Iteration 359 controller gates, OUTSIDE the codex sandbox, sequential (plan: "Run suites and
# mutations sequentially"). A2, A3 (collection; review by the controller), A4, A6, A7, A8 in the
# sprint worktree; then A5 (mutation runner) in its own detached worktree at the same commit.
set -u
WT=/Users/voightkampff/dev/sunholo-data/.wt-world-iter359-sprint
MW=/Users/voightkampff/dev/sunholo-data/.mut-wt-iter359
cd "$WT" || exit 2
export AILANG_BIN="$HOME/.pinned-ailang/ailang"
export EVIDENCE="$WT/design_docs/verification/world-iter359/controller-M1"
export TMPDIR=/Users/voightkampff/dev/sunholo-data/.ctl-tmp-iter359
mkdir -p "$EVIDENCE" "$TMPDIR"
S="$EVIDENCE/summary.txt"
note() { echo "$*" | tee -a "$S"; }
note "# iteration 359 controller gates on $(git rev-parse HEAD) start $(date -u +%FT%TZ) load $(sysctl -n vm.loadavg)"
note "go: $(go version); node: $(node --version); ailang: $("$AILANG_BIN" --version | head -1)"

# A1 hashes still hold
shasum -a 256 -c > "$EVIDENCE/A1-hashes.txt" 2>&1 <<'SHA'
4efb18438d736bb41a0770b3d4fd0b12e3ff5360e89dc8bc156e2ab44f67521f  .github/workflows/ci.yml
110fffc3f31a10d71a024b3e02f66b0864178ce59e44e3e5b5be2c82ac11c79f  host/broker/exec_probe.go
8c7f9602c1105bdca99af2326a6e51851d7011e1ccb875f18a293f4f91838b0e  host/broker/exec_probe_net6_dial_test.go
8ec970891599257aa280436f78cfac427ebaccc1cf5cd33d404904cf60ec3b26  host/broker/exec_probe_net6_test.go
ee59ba2cc5742855f6aa172cb7873484a5c07abf2f1b73d5792d8673c45e7366  host/broker/exec_probe_test.go
SHA
note "A1 post-apply hashes rc=$?"

# A2
gofmt -l host/broker/exec_probe.go host/broker/exec_probe_net6_dial_test.go host/broker/exec_probe_net6_test.go host/broker/exec_probe_test.go > "$EVIDENCE/gofmt.log"; note "A2 gofmt-files-listed=$(wc -l < "$EVIDENCE/gofmt.log" | tr -d ' ')"
go vet ./host/broker/ > "$EVIDENCE/A2-vet.log" 2>&1; note "A2 vet rc=$?"
go build ./... > "$EVIDENCE/A2-build.log" 2>&1; note "A2 build rc=$?"
/bin/bash -n design_docs/verification/world-iter359/plan-mutants.sh.txt; note "A2 runner bash -n rc=$?"

# A4
fails=0
for name in TestExecProbePassThroughShimFailsEveryConfinementArm TestExecProbeNet6ForeignDialIsRetriedNotRefused TestExecProbeNet6ChildOrSilentAcceptStillFails TestExecProbeNet6ForeignEveryAttemptRefuses TestExecProbeNet6DialClassifier TestExecProbeNet6BoundBeforeAnyPick TestExecProbeNet6BindSiteIsBeforeTheFanOut TestExecProbeReverseLegJudge TestExecProbeReverseLegChattyForeignHolderIsRetried TestExecProbeReverseLegExhaustionRefusesDistinctly TestExecProbeReverseLegRealReachSurvivesAForeignAttempt TestExecProbeReverseLegNoTokenIsNotRetried TestExecProbeForeignListenerOnTheReversePortDoesNotRefuse TestExecProbeReverseLegSilentHolderRefusesSticky TestExecProbeReverseDialErrIsSilent TestExecProbeReversePollRecordsConnectReset TestExecProbeReverseHelloWaitBounds TestExecProbeReversePollCadence TestExecProbeChildListenerDrainsTheHello; do
  log="$EVIDENCE/A4-$name.log"; rc=0
  go test -count=1 -run "^(${name})$" ./host/broker/ -v -timeout 300s > "$log" 2>&1 || rc=$?
  p=$(grep -c -- "^--- PASS: $name " "$log"); sk=$(grep -c -- '--- SKIP:' "$log")
  [ "$rc" -eq 0 ] && [ "$p" -eq 1 ] && [ "$sk" -eq 0 ] || fails=$((fails+1))
  note "A4 $name rc=$rc pass=$p skip=$sk"
done
note "A4 failures=$fails"

# A6
go test -count=3 -run '^(TestExecProbePassThroughShimFailsEveryConfinementArm|TestExecProbeNet6.*)$' -v ./host/broker/ > "$EVIDENCE/A6-stability.log" 2>&1; rc=$?
note "A6 rc=$rc $(grep -E '^--- (PASS|FAIL|SKIP)' "$EVIDENCE/A6-stability.log" | sed -E 's/ \([0-9.]+s\)//' | sort | uniq -c | tr '\n' ';')"

# A7
go test -count=1 -timeout 600s ./host/broker/ > "$EVIDENCE/A7-broker.log" 2>&1; note "A7 broker rc=$? $(tail -1 "$EVIDENCE/A7-broker.log")"
./scripts/verify_ail.sh > "$EVIDENCE/A7-verify-ail.log" 2>&1; note "A7 verify_ail rc=$? $(tail -1 "$EVIDENCE/A7-verify-ail.log")"
go vet ./host/broker/ >/dev/null 2>&1; note "A7 vet rc=$?"
./scripts/verify_go.sh > "$EVIDENCE/A7-verify-go.log" 2>&1; note "A7 verify_go rc=$? $(tail -1 "$EVIDENCE/A7-verify-go.log")"

# A8
python3 -I - > "$EVIDENCE/A8.log" 2>&1 <<'PYCODE'
from pathlib import Path
s=Path('.github/workflows/ci.yml').read_text()
a=s.index('      - name: Row 140 Workspace.Exec real-srt tests, verbose (linux bubblewrap)')
b=s.index('      - name:',a+1); block=s[a:b]
lines=[l for l in block.splitlines() if l.lstrip().startswith('tests=')]
assert len(lines)==4
names=('TestExecProbeNet6ForeignDialIsRetriedNotRefused','TestExecProbeNet6ChildOrSilentAcceptStillFails','TestExecProbeNet6ForeignEveryAttemptRefuses','TestExecProbeNet6DialClassifier')
assert 'TestExecSrtStartupProbePasses' in lines[1]
for n in names:
    assert sum(l.count(n) for l in lines)==1,n
    assert n in lines[2],n
for text in ('WORLD_EXEC_SRT_NODE_MODULES','RUNNER_OS','go test ./host/broker/ ./host/daemon/','for t in $tests; do','^--- PASS: $t '):
    assert text in block,text
print('A8 ok')
PYCODE
note "A8 rc=$? $(tail -1 "$EVIDENCE/A8.log")"
note "tree-status-after-gates: $(git status --porcelain | grep -v 'design_docs/verification/world-iter359/controller-M1' | wc -l | tr -d ' ') non-evidence changes"

# A5 in its own detached worktree
git worktree add --detach "$MW" HEAD > /dev/null 2>&1; note "A5 mutation worktree at $(git -C "$MW" rev-parse --short HEAD)"
bash "$MW/design_docs/verification/world-iter359/plan-mutants.sh.txt" "$MW" > "$EVIDENCE/A5-mutants.log" 2>&1; rc=$?
note "A5 runner rc=$rc match=$(grep -c '^EXPECTATION MATCH:' "$EVIDENCE/A5-mutants.log") mismatch=$(grep -c '^EXPECTATION MISMATCH:' "$EVIDENCE/A5-mutants.log") restore-ok=$(grep -c '^RESTORE cmp ok: .*tests=' "$EVIDENCE/A5-mutants.log") census=$(grep -c 'GO FILE CENSUS unchanged; no extra .go files remain' "$EVIDENCE/A5-mutants.log")"
note "A5 mutation-tree status: $(git -C "$MW" status --porcelain | wc -l | tr -d ' ') changes"
note "# end $(date -u +%FT%TZ)"
echo GATES-DONE
