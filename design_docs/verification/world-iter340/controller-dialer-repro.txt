import (
	"fmt"
	"net"
	"sync/atomic"
	"time"
)

func main() {
	var total atomic.Int64
	const n = 16
	lns := make([]net.Listener, n)
	for i := range lns {
		ln, err := net.Listen("tcp", "127.0.0.1:0")
		if err != nil {
			panic(err)
		}
		lns[i] = ln
		go func() {
			for {
				c, err := ln.Accept()
				if err != nil {
					return
				}
				total.Add(1)
				fmt.Printf("%s accept on %s from %s\n", time.Now().Format("15:04:05.000"), ln.Addr(), c.RemoteAddr())
				c.Close()
			}
		}()
	}
	fmt.Printf("holding %d listeners for 75s from %s\n", n, time.Now().Format("15:04:05"))
	time.Sleep(75 * time.Second)
	fmt.Printf("TOTAL unsolicited accepts=%d over %d listeners\n", total.Load(), n)
}
