package broker

import (
	"fmt"
	"net"
	"testing"
	"time"
)

// SCRATCH-ONLY (row 242 designer r2): who dials idle loopback listeners. The
// instrumented accept loop prints lsof for every accept after the first.
func TestZZR2Attribute(t *testing.T) {
	var ls []*probeListener
	for i := 0; i < 4; i++ {
		l := newProbeListener()
		if c, err := net.Dial("tcp", fmt.Sprintf("127.0.0.1:%d", l.port)); err == nil { // accept #1 is ours
			_ = c.Close()
		}
		ls = append(ls, l)
	}
	time.Sleep(65 * time.Second)
	for _, l := range ls {
		fmt.Printf("ZZR2B port=%d accepts=%d\n", l.port, l.accepts.Load())
		l.close()
	}
}
