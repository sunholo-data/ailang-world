package broker

import (
	"context"
	"fmt"
	"os"
	"regexp"
	"testing"
	"time"
)

// SCRATCH-ONLY (row 242 designer): every failure path's per-profile text, normalised.
func TestZZEquiv(t *testing.T) {
	norm := func(s string) string {
		s = regexp.MustCompile(`wep-[0-9a-f]{16}`).ReplaceAllString(s, "wep-X")
		s = regexp.MustCompile(`127\.0\.0\.1:\d+`).ReplaceAllString(s, "127.0.0.1:P")
		s = regexp.MustCompile(`/(private/)?var/folders/[^ "]*?/T/TestZZEquiv[^/]*/`).ReplaceAllString(s, "TMP/")
		return s
	}
	for _, row := range []struct {
		name string
		fail bool
	}{{"bind-ok", false}, {"bind-fails", true}} {
		t.Run(row.name, func(t *testing.T) {
			f := newProbeFixture(t, shimSrtTree(t, "passthrough"), nil, "", []any{"sh", "-c", "echo toolchain-ok"})
			bad := mustParseProfile(t, map[string]any{"profile": ExecProfileVersion, "project": "badroot",
				"path": []any{"/usr/bin", "/bin"}, "timeout_ms": 9000, "probe": []any{"sh", "-c", "exit 0"},
				"read_roots": []any{"/nonexistent-zz-iter340"}, "commands": map[string]any{"true": map[string]any{"argv": []any{"sh", "-c", "exit 0"}}}})
			zzFailListen.Store(row.fail)
			defer zzFailListen.Store(false)
			before := loopbackListenPorts()
			ctx, cancel := context.WithTimeout(context.Background(), 60*time.Second)
			defer cancel()
			errs, err := ProbeExecProfilesEach(ctx, ExecProbeConfig{Sandbox: f.sandbox, Paths: f.paths}, []*ExecProfile{f.profile, bad})
			fmt.Fprintf(os.Stdout, "ZZ %s top-err=%v n=%d\n", row.name, err, len(errs))
			for i, e := range errs {
				fmt.Fprintf(os.Stdout, "ZZ %s errs[%d] %T: %s\n", row.name, i, e, norm(fmt.Sprint(e)))
			}
			time.Sleep(300 * time.Millisecond)
			left := 0
			for p := range loopbackListenPorts() {
				if !before[p] {
					left++
				}
			}
			fmt.Fprintf(os.Stdout, "ZZ %s listeners-left=%d\n", row.name, left)
			f.assertProbeCleaned(t)
		})
	}
}
