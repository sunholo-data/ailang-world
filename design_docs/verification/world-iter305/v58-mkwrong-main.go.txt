// Builds the deliberately wrong phone crops for V58: a timeline-only crop
// (rows shifted up by 330 px, bottom filled with page background) and a blank frame.
package main

import (
	"image"
	"image/color"
	"image/draw"
	"image/png"
	"os"
)

func main() {
	f, _ := os.Open(os.Args[1])
	src, err := png.Decode(f)
	if err != nil {
		panic(err)
	}
	b := src.Bounds()
	bg := src.At(5, 330)
	w := image.NewNRGBA(b)
	draw.Draw(w, b, &image.Uniform{bg}, image.Point{}, draw.Src)
	draw.Draw(w, image.Rect(0, 0, b.Dx(), b.Dy()-330), src, image.Pt(0, 330), draw.Src)
	o, _ := os.Create(os.Args[2])
	png.Encode(o, w)
	o.Close()
	bl := image.NewNRGBA(b)
	draw.Draw(bl, b, &image.Uniform{color.NRGBA{250, 250, 250, 255}}, image.Point{}, draw.Src)
	o2, _ := os.Create(os.Args[3])
	png.Encode(o2, bl)
	o2.Close()
}
