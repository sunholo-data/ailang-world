+// T12 (row 241 design §2.1): pollLoopback's REAL dial cadence is
+// execProbePollEvery, observed through the dial seam, so T10's ceiling (the
+// listen window less one poll period) derives from the interval the poll
+// actually waits. The clock is a reference loop on this goroutine that waits
+// execProbePollEvery revCadencePeriods times, the same timer the poll should
+// run, so load that delays the poll's wake-ups delays the reference alike.
+//   - count: the poll dials once at once and once per period, so it makes
+//     about revCadencePeriods+1 dials while the reference runs; fewer than
+//     half of that means its wait is over twice execProbePollEvery.
+//   - floor: every gap between dials is at least execProbePollEvery. A Go
+//     timer never fires early and the next dial follows it, so load cannot
+//     red this check.
+func TestExecProbeReversePollCadence(t *testing.T) {
+	const revCadencePeriods = 10
+	refused := revDialErr(syscall.ECONNREFUSED)
+	var mu sync.Mutex
+	var at []time.Time
+	stop := pollLoopback(40000, "WORLD-PROBE-NONCE "+strings.Repeat("a", 32), func(string) (net.Conn, error) {
+		mu.Lock()
+		at = append(at, time.Now())
+		mu.Unlock()
+		return nil, refused
+	})
+	start := time.Now()
+	for range revCadencePeriods {
+		<-time.After(execProbePollEvery)
+	}
+	span := time.Since(start)
+	stop()
+	mu.Lock()
+	defer mu.Unlock()
+	if len(at) < revCadencePeriods/2+1 {
+		t.Errorf("%d dials while a reference waited execProbePollEvery (%v) %d times (%v); want at least %d: the poll does not wait execProbePollEvery", len(at), execProbePollEvery, revCadencePeriods, span, revCadencePeriods/2+1)
+	}
+	fastest, slowest := time.Duration(0), time.Duration(0)
+	for i := 1; i < len(at); i++ {
+		gap := at[i].Sub(at[i-1])
+		if i == 1 || gap < fastest {
+			fastest = gap
+		}
+		slowest = max(slowest, gap)
+		if gap < execProbePollEvery {
+			t.Errorf("dial %d followed dial %d after %v; want no gap under execProbePollEvery (%v)", i+1, i, gap, execProbePollEvery)
+		}
+	}
+	t.Logf("CADENCE dials=%d span=%v fastest=%v slowest=%v", len(at), span, fastest, slowest)
+}
