#!/usr/bin/env python3
# Worst lag of the poll behind the reference across T12 v2/v3 runs: ratio = periods / (dials - 1),
# i.e. reference periods per poll gap (1.0 = lockstep; the v3 gate admits up to 20/5 = 4.0, v2 up to 10/5 = 2.0).
import re,sys
for path,periods in [(a.split(':')[0], int(a.split(':')[1])) for a in sys.argv[1:]]:
    n=[int(x) for x in re.findall(r"CADENCE dials=(\d+) ", open(path).read())]
    worst=max(periods/(x-1) if x>1 else float('inf') for x in n)
    print(f"{path}: runs={len(n)} min dials={min(n)} worst lag ratio={worst:.2f} (gate admits {periods/5:.1f})")
