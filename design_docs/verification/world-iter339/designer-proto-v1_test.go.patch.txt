+// T12 (row 241 design §2.1): pollLoopback's REAL dial cadence is
+// execProbePollEvery, measured through the dial seam, so T10's ceiling (the
+// listen window less one poll period) derives from the interval the poll
+// actually waits. Two checks, both on the gaps between consecutive dials:
+//   - floor: every gap is at least execProbePollEvery. A Go timer never fires
+//     early and the next dial follows it, so load cannot red this check.
+//   - ceiling: the FASTEST gap is under two periods. Load only lengthens a
+//     gap, and the minimum over up to revCadenceGaps gaps needs every wake-up
+//     late by a whole period to red a correct tree.
+//
+// The collection window is one listen hold: a poll that cannot dial twice
+// while the child holds its listener cannot see it either.
+func TestExecProbeReversePollCadence(t *testing.T) {
+	const revCadenceGaps = 8
+	refused := revDialErr(syscall.ECONNREFUSED)
+	listenWindow := time.Duration(execProbeListenMS) * time.Millisecond
+	dials := make(chan time.Time, revCadenceGaps+1)
+	stop := pollLoopback(40000, "WORLD-PROBE-NONCE "+strings.Repeat("a", 32), func(string) (net.Conn, error) {
+		select {
+		case dials <- time.Now():
+		default: // the test has what it needs; later dials are not recorded
+		}
+		return nil, refused
+	})
+	var at []time.Time
+	window := time.After(listenWindow)
+collect:
+	for len(at) < revCadenceGaps+1 {
+		select {
+		case d := <-dials:
+			at = append(at, d)
+		case <-window:
+			break collect
+		}
 	}
+	stop()
+	if len(at) < 2 {
+		t.Fatalf("%d dial(s) in one listen window (%v); want at least 2 to measure the cadence", len(at), listenWindow)
+	}
+	fastest, slowest := at[1].Sub(at[0]), at[1].Sub(at[0])
+	for i := 1; i < len(at); i++ {
+		gap := at[i].Sub(at[i-1])
+		fastest, slowest = min(fastest, gap), max(slowest, gap)
+		if gap < execProbePollEvery {
+			t.Errorf("dial %d followed dial %d after %v; want no gap under execProbePollEvery (%v)", i+1, i, gap, execProbePollEvery)
+		}
+	}
+	if fastest >= 2*execProbePollEvery {
+		t.Errorf("fastest of %d dial gaps %v; want under two poll periods (%v): the poll does not wait execProbePollEvery", len(at)-1, fastest, 2*execProbePollEvery)
+	}
+	t.Logf("CADENCE gaps=%d fastest=%v slowest=%v excess=%v", len(at)-1, fastest, slowest, fastest-execProbePollEvery)
 }
