// daemon-construction audit (iteration 327, controller): every place a value of
// type host/daemon.Daemon comes into being, resolved by go/types (aliases included).
package main

import (
	"fmt"
	"go/ast"
	"go/types"
	"os"
	"sort"

	"golang.org/x/tools/go/packages"
)

const target = "github.com/sunholo-data/ailang-world/host/daemon.Daemon"

func isDaemon(t types.Type) bool {
	if t == nil {
		return false
	}
	n, ok := types.Unalias(t).(*types.Named)
	return ok && n.Obj().Pkg() != nil && n.Obj().Pkg().Path()+"."+n.Obj().Name() == target
}

func main() {
	cfg := &packages.Config{Mode: packages.NeedName | packages.NeedFiles | packages.NeedSyntax | packages.NeedTypes | packages.NeedTypesInfo, Tests: true, Dir: os.Args[1]}
	pkgs, err := packages.Load(cfg, "./...")
	if err != nil {
		fmt.Println("LOAD ERROR", err)
		os.Exit(2)
	}
	seen := map[string]bool{}
	var out []string
	errs := 0
	for _, p := range pkgs {
		errs += len(p.Errors)
		for _, f := range p.Syntax {
			ast.Inspect(f, func(n ast.Node) bool {
				var kind string
				var pos ast.Node
				switch x := n.(type) {
				case *ast.CompositeLit:
					if isDaemon(p.TypesInfo.TypeOf(x)) {
						kind, pos = "composite-literal", x
					}
				case *ast.CallExpr:
					if id, ok := x.Fun.(*ast.Ident); ok && id.Name == "new" && len(x.Args) == 1 && isDaemon(p.TypesInfo.TypeOf(x.Args[0])) {
						kind, pos = "new()", x
					}
				case *ast.ValueSpec:
					if len(x.Values) == 0 {
						for _, nm := range x.Names {
							if isDaemon(p.TypesInfo.TypeOf(nm)) {
								kind, pos = "zero-value var", x
							}
						}
					}
				}
				if kind != "" {
					s := fmt.Sprintf("%s\t%s", p.Fset.Position(pos.Pos()), kind)
					if !seen[s] {
						seen[s] = true
						out = append(out, s)
					}
				}
				return true
			})
		}
	}
	sort.Strings(out)
	for _, s := range out {
		fmt.Println(s)
	}
	fmt.Printf("packages=%d load-errors=%d constructions=%d\n", len(pkgs), errs, len(out))
}
