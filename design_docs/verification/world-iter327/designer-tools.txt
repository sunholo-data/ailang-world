# usage: mut.py FILE LINE OLD NEW  -- replace OLD with NEW exactly once on LINE (1-indexed)
import sys
path, line, old, new = sys.argv[1], int(sys.argv[2]), sys.argv[3], sys.argv[4]
lines = open(path).read().split('\n')
src = lines[line-1]
assert src.count(old) == 1, (line, src, old)
lines[line-1] = src.replace(old, new)
open(path, 'w').write('\n'.join(lines))

import sys, collections
# usage: covsum.py FILE_SUFFIX LINE... < profiles ; sums counts of every block whose [start,end] line span contains LINE
suffix, lines = sys.argv[1], [int(x) for x in sys.argv[2:]]
blocks = collections.defaultdict(int)
for raw in sys.stdin:
    if raw.startswith('mode:') or suffix not in raw: continue
    loc, nstmt, cnt = raw.rsplit(' ', 2)
    blocks[loc] += int(cnt)
for ln in lines:
    hits = [(loc, c) for loc, c in blocks.items() if int(loc.split(':')[1].split('.')[0]) <= ln <= int(loc.split(',')[1].split('.')[0])]
    for loc, c in sorted(hits): print(f"line {ln}: block {loc.split('/')[-1]} count={c}")
    if not hits: print(f"line {ln}: no block")
