// scriptConn is one scripted connected dial: Write returns werr; Read hands
// over answer one byte per call, then returns rerr (io.EOF when nil).
type scriptConn struct {
	net.Conn
	answer []byte
	werr   error
	rerr   error
}

func (c *scriptConn) SetDeadline(time.Time) error     { return nil }
func (c *scriptConn) SetReadDeadline(time.Time) error { return nil }
func (c *scriptConn) Close() error                    { return nil }
func (c *scriptConn) Write(b []byte) (int, error) {
	if c.werr != nil {
		return 0, c.werr
	}
	return len(b), nil
}
func (c *scriptConn) Read(b []byte) (int, error) {
	if len(c.answer) > 0 {
		n := copy(b, c.answer[:1])
		c.answer = c.answer[n:]
		return n, nil
	}
	if c.rerr != nil {
		return 0, c.rerr
	}
	return 0, io.EOF
}

func zzOpErr(op string, err error) error {
	return &net.OpError{Op: op, Net: "tcp", Err: os.NewSyscallError(op, err)}
}

// scriptedDial returns the steps in order, then ECONNREFUSED; all is closed
// once every step has been handed out, so the test stops the poll only after
// the whole script ran (no sleep: the poll's own 50 ms cadence paces it).
func scriptedDial(steps []byte, want string) (func(string) (net.Conn, error), chan struct{}) {
	all := make(chan struct{})
	var once sync.Once
	i := 0
	return func(string) (net.Conn, error) {
		if i >= len(steps) {
			once.Do(func() { close(all) })
			return nil, revDialErr(syscall.ECONNREFUSED)
		}
		s := steps[i]
		i++
		if i == len(steps) {
			defer once.Do(func() { close(all) })
		}
		switch s {
		case 'N':
			return &scriptConn{answer: []byte(want + "\n")}, nil
		case 'F':
			return &scriptConn{answer: []byte("HTTP/1.1 400 Bad Request\r\n")}, nil
		case 'f':
			return &scriptConn{}, nil // clean EOF after the hello
		case 'S':
			return &scriptConn{rerr: zzOpErr("read", syscall.ECONNRESET)}, nil
		case 'W':
			return &scriptConn{werr: zzOpErr("write", syscall.ECONNRESET)}, nil // read then EOF
		case 'D':
			return &scriptConn{rerr: &net.OpError{Op: "read", Net: "tcp", Err: os.ErrDeadlineExceeded}}, nil
		case 'C':
			return nil, revDialErr(syscall.ECONNRESET)
		case 'A':
			return nil, revDialErr(syscall.ECONNABORTED)
		case 'T':
			return nil, &net.OpError{Op: "dial", Net: "tcp", Err: os.ErrDeadlineExceeded}
		case 'P': // a read reset observed at least execProbeListenMS after this dial began
			time.Sleep(time.Duration(execProbeListenMS) * time.Millisecond)
			return &scriptConn{rerr: zzOpErr("read", syscall.ECONNRESET)}, nil
		case 'L', 'G': // a foreign answer whose dial took at least 600 ms
			time.Sleep(600 * time.Millisecond)
			return &scriptConn{answer: []byte("HTTP/1.1 400 Bad Request\r\n")}, nil
		case 'Q': // a read reset whose dial took at least 450 ms
			time.Sleep(450 * time.Millisecond)
			return &scriptConn{rerr: zzOpErr("read", syscall.ECONNRESET)}, nil
		}
		panic("bad step")
	}, all
}

func TestZZIter330HolderExitAfterForeign(t *testing.T) {
	want := "WORLD-PROBE-NONCE " + strings.Repeat("b", 32)
	fs, f, s, c := reverseSeen{foreign: true, silent: true}, reverseSeen{foreign: true}, reverseSeen{silent: true}, reverseSeen{child: true, foreign: true}
	for _, row := range []struct {
		name  string
		steps string
		clean bool
		seen  reverseSeen
		v     revVerdict
	}{
		{"FS", "FS", true, f, revInconclusive},
		{"FC", "FC", true, f, revInconclusive},
		{"FW", "FW", true, f, revInconclusive},
		{"fS", "fS", true, f, revInconclusive},
		{"GGFS", "GGFS", true, f, revInconclusive},
		{"FD", "FD", true, fs, revSilent},
		{"SF", "SF", true, fs, revSilent},
		{"CF", "CF", true, fs, revSilent},
		{"FSD", "FSD", true, fs, revSilent},
		{"S", "S", true, s, revSilent},
		{"FP", "FP", true, fs, revSilent},
		{"LQ", "LQ", true, fs, revSilent},
		{"FS-unclean", "FS", false, fs, revSilent},
		{"FSN", "FSN", true, c, revReached},
		{"FA", "FA", true, f, revInconclusive},
		{"A", "A", true, reverseSeen{}, revPass},
		{"FT", "FT", true, f, revInconclusive},
		{"T", "T", true, reverseSeen{}, revPass},
	} {
		t.Run(row.name, func(t *testing.T) {
			dial, all := scriptedDial([]byte(row.steps), want)
			stop := pollLoopback(40000, want, dial)
			select {
			case <-all:
			case <-time.After(20 * time.Second):
				t.Fatal("the poll did not run the script within 20 s")
			}
			seen := stop(row.clean)
			if v := judgeReverse("WORLD-PROBE-BOUND", seen); seen != row.seen || v != row.v {
				t.Fatalf("dials %s (clean %v): seen %+v verdict %d; want %+v verdict %d", row.steps, row.clean, seen, v, row.seen, row.v)
			}
		})
	}
}
