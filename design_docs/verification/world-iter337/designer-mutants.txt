### mutate.py
import sys, re
# usage: mutate.py <worktree> <mutant-id>   (applies exactly one edit; asserts its anchor occurs once)
root, mid = sys.argv[1], sys.argv[2]
R = root + '/host/workbench/render.go'
P = root + '/cmd/ailang-worldd/zz_iter337_still_proto_test.go'
LIVE = '{{if .Still}}{{else}}<script src="/workbench/live.js" defer></script>{{end}}'
def edit(path, old, new):
    s = open(path).read()
    assert s.count(old) == 1, (mid, path, s.count(old))
    open(path, 'w').write(s.replace(old, new))
if mid == 'MS1-still-keeps-livejs':
    edit(R, LIVE, '<script src="/workbench/live.js" defer></script>')
elif mid == 'MS2-extra-script-both-modes':
    edit(R, LIVE, LIVE + '<script type="application/json">{}</script>')
elif mid == 'MS3-still-constant-marker':
    # html/template strips comments, so the marker is an element (a <!--still--> variant was a no-op: PASS)
    edit(R, LIVE, '{{if .Still}}<meta name="still" content="1">{{else}}<script src="/workbench/live.js" defer></script>{{end}}')
elif mid == 'MS4-still-varies-after-1s':
    edit(R, 'import (\n\t"errors"\n', 'import (\n\t"errors"\n\t"time"\n')
    edit(R, 'func Render(w io.Writer, p Page) error { return pageTemplate.Execute(w, p) }',
         'var stillFirst time.Time\n\nfunc Render(w io.Writer, p Page) error {\n\tif p.Still {\n\t\tif stillFirst.IsZero() {\n\t\t\tstillFirst = time.Now()\n\t\t}\n\t\tif time.Since(stillFirst) > time.Second {\n\t\t\tp.Title += " "\n\t\t}\n\t}\n\treturn pageTemplate.Execute(w, p)\n}')
elif mid == 'MS5-livejs-removed-everywhere':
    edit(R, LIVE, '')
elif mid == 'MT1-routes-nil':
    edit(P, 'routes := marketingStillRoutes(t, check, rig.manifest)\n', 'routes := marketingStillRoutes(t, check, rig.manifest)[:0]\n')
elif mid == 'MT2-invert-repeat':
    edit(P, '\t\tif !repeat {', '\t\tif repeat {')
elif mid == 'MT3-drop-minus-tag':
    edit(P, '\t\tif !stripped {', '\t\tif false && !stripped {')
elif mid == 'MT4-drop-sleep':
    edit(P, '\t\ttime.Sleep(2 * time.Second)\n', '')
elif mid == 'MT5-drop-default-control':
    edit(P, '\t\tif tags != 1 {', '\t\tif false && tags != 1 {')
else:
    sys.exit('unknown mutant ' + mid)
print('applied', mid)
### run.sh

wt=$1; label=$2; shift 2
