   128	async def _stream_with_first_token_deadline(stream: Any, deadline_s: float, model: str):
   129	    """Yield from ``stream``; if its FIRST item takes longer than ``deadline_s``,
   130	    close it and raise ``SilentStallError``. Once anything has arrived the
   131	    stream runs to completion however long the rest takes — a slow tail is
   132	    not a stall."""
   133	    iterator = stream.__aiter__()
   134	    try:
   135	        first = await asyncio.wait_for(iterator.__anext__(), timeout=deadline_s)
   136	    except StopAsyncIteration:
   137	        return
   138	    except TimeoutError:
   139	        aclose = getattr(iterator, "aclose", None)
   140	        if aclose is not None:
   141	            try:
   142	                await aclose()
   143	            except BaseException as exc:  # a dead stream may object to closing; irrelevant now
   144	                logger.debug("first-token: aclose after timeout raised %r (ignored)", exc)
   145	        raise SilentStallError(deadline_s, model) from None
   146	    yield first
   147	    async for item in iterator:
   148	        yield item
