# Usage: bash /absolute/path/plan-mutants.sh.txt TREE_ROOT
# Sequential execution only; no concurrent source edits or test processes.
# Runner artifacts/caches stay under TREE_ROOT; existing test t.TempDir() calls
# still use the unchanged process TMPDIR (plan finding F1).

export GOPATH="$state/runtime/gopath"  # controller copy, iteration 360: GOTMPDIR export removed (see controller-M1/A5-instrument.txt)

restore() {
  while IFS= read -r rel; do
    cp "$state/pristine/$rel.pristine" "$root/$rel" || return 1
    cmp "$state/pristine/$rel.pristine" "$root/$rel" || return 1
  done < "$state/paths.txt"
}
finish() {
  saved_rc=$?
  trap - EXIT HUP INT TERM
  if restore; then echo 'RESTORE cmp ok: exit cleanup'; else saved_rc=3; fi
  chmod -R u+w "$state/runtime" 2>/dev/null || true
  rm -rf "$state/runtime" "$state/pristine" || saved_rc=3
  if ! python3 -I - "$root" "$state/go-before.json" <<'CENSUS'

print('GO FILE CENSUS unchanged; no extra .go files remain')
CENSUS
  then saved_rc=3; fi
  echo "Mutation evidence: $state (logs retained; pristine and runtime removed)"
  exit "$saved_rc"
}

run_row() { # mutant package tests expected_rc label
  mut=$1; pkg=$2; names=$3; expected=$4; label=$5
  rows=$((rows+1))
  restore
  mutate "$mut"
  log="$state/row-$rows.jsonl"
  rc=0
  go test -count=1 -run "^($names)$" -json -timeout 300s "./$pkg/" > "$log" 2>&1 || rc=$?
  # Reject build/setup failures, absent tests and SKIP; every named identity
  # must end in the expected state. An unrelated failure cannot kill a mutant.
  valid=0
  python3 -I - "$log" "$names" "$expected" <<'CHECK' || valid=$?

import collections,json,sys
log,names,expected=Path(sys.argv[1]),sys.argv[2].split('|'),int(sys.argv[3])

except Exception as exc: raise SystemExit('invalid go-test JSON: '+str(exc))
want='fail' if expected else 'pass'
terminal=collections.defaultdict(list)
for e in events:
    test=e.get('Test','')
    if test in names and e.get('Action') in ('pass','fail','skip'): terminal[test].append(e['Action'])
if any(terminal[n]!=[want] for n in names): raise SystemExit('named terminal mismatch: '+str(dict(terminal)))
if not any(e.get('Action')==want and not e.get('Test') for e in events): raise SystemExit('missing package terminal')
CHECK
  if [ "$rc" -eq "$expected" ] && [ "$valid" -eq 0 ]; then
    echo "EXPECTATION MATCH: $mut package=$pkg tests=$names label=$label expected_rc=$expected rc=$rc"
  else
    echo "EXPECTATION MISMATCH: $mut package=$pkg tests=$names label=$label expected_rc=$expected rc=$rc identities_rc=$valid"
    cat "$log"
    mismatches=$((mismatches+1))
  fi
  restore
  echo "RESTORE cmp ok: $mut package=$pkg tests=$names"
}
