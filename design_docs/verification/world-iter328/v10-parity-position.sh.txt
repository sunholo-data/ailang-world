P=${P:-/tmp/w328d/banked/p0check/p0check}; MC=${MC:-/tmp/w328d/mc}; cd /tmp/w328d
args=("r1-seed1/own-rangebox=$V/v64-rects-r1-seed1.json" "r1-seed2/own-rangebox=$V/v64-rects-r1-seed2.json" "r1-seed1/block-converted=$V/v64-rects-r1-seed1-blockconverted.json" "r1-seed2/block-converted=$V/v64-rects-r1-seed2-blockconverted.json")

import sys,re
def parse(p, banked):
    out={}; cur=None
    for line in open(p):
        if line.startswith('TOTAL') or line.startswith('rc='): continue
        if not line.startswith(' '):
            f=line.split(); cur=f[0]
            out[cur]=[(f[2] if banked else f[1]),[]]
        else:
            t=line.strip()
            if banked:
                if t.startswith('r3: '): out[cur][1].append(t[4:])
            else: out[cur][1].append(t)
    return out
a=parse(sys.argv[1],True); b=parse(sys.argv[2],False)
assert a.keys()==b.keys(), (a.keys()^b.keys())
n=len(a); vd=sum(a[k][0]!=b[k][0] for k in a); rd=sum(a[k][1]!=b[k][1] for k in a)
acc=sum(a[k][0]=='ACCEPT' for k in a)
print(f"PARITY position: inputs={n} banked-accept={acc} verdict-diffs={vd} reason-diffs={rd}")
for k in a:
    if a[k]!=b[k]: print("DIFF",k,a[k],b[k])
PY
