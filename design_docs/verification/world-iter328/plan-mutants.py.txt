func mutStd(im image.Image, r Rect) float64 {
	var s, ss, n float64
	for y := r.Y; y < r.Y+r.H; y++ {
		for x := r.X; x < r.X+r.W; x++ {
			v := phoneLuma(im, x, y)
			s += v
			ss += v * v
			n++
		}
	}
	if n == 0 {
		return 0
	}
	m := s / n
	return math.Sqrt(ss/n - m*m)
}
'''

env = dict(os.environ, AILANG_BIN=os.path.expanduser("~/.pinned-ailang/ailang"))
for row in MUTANTS:
    mid, rel, old, new, regex = row[:5]
    pkg = row[5] if len(row) > 5 else "./" + MC
    if only and mid not in only:
        continue
    path = os.path.join(T, rel)
    orig = open(path, "rb").read()
    s = orig.decode()
    if old is None:
        mutated = s + new
    else:
        if s.count(old) != 1:
            print(f"{mid}: SETUP-ERROR {old[:60]!r} matches {s.count(old)}"); continue
        mutated = s.replace(old, new)
        if mid == "F3":
            mutated = mutated.replace('import (\n', 'import (\n\t"math"\n', 1) + STD
    open(path, "w").write(mutated)
    t0 = time.time()
    r = subprocess.run(["go", "test", "-count=1", "-timeout", "20m", "-v", "-run", regex, pkg], cwd=T, env=env, capture_output=True, text=True)
    dt = time.time() - t0
    open(path, "wb").write(orig)
    assert open(path, "rb").read() == orig
    out = r.stdout + r.stderr
    fails = re.findall(r"^\s*--- FAIL: (\S+)", out, re.M)
    leaf = [f for f in fails if not any(g.startswith(f + "/") for g in fails)]
    first = ""
    m = re.search(r"^\s+\S+_test\.go:\d+: (?!PARITY|subprocess-timeout after|unminted seed refusal|WORLD_PHONE_CORPUS unset)(.*)$", out, re.M)
    if m: first = m.group(1)[:160]
    build = "BUILD-FAILED " if "[build failed]" in out or "build failed" in out else ""
    verdict = "KILLED" if r.returncode != 0 and fails else ("BUILD-FAILED" if build else "SURVIVED")
    print(f"{mid}: {verdict} rc={r.returncode} {dt:.1f}s red={len(leaf)} {' '.join(leaf)}")
    if first: print(f"    first: {first}")
    sys.stdout.flush()
