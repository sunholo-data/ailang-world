// Iteration-328 designer: phone arms that isolate (ii) (absent from V67, whose (ii)-failing arms
// also fail (i)). usage: newarms <phone.png> <outdir>
//   header-solid-ink.png  header band rows 45..67, x 25..369 painted luma 0: bands, (i), title clip
//                         unchanged; the header region's std-dev is 0 (fails (ii) alone)
//   tall-845.png          one page-colour row appended (390x845): fails only (iii′) dims
//   entry-solid-ink.png   first-entry union rows 428..609, x 41..363 painted luma 0 (fails (ii) alone)
package main

import (
	"image"
	"image/color"
	"image/draw"
	"image/png"
	"os"
	"path/filepath"
)

func main() {
	f, _ := os.Open(os.Args[1])
	src, err := png.Decode(f)
	if err != nil {
		panic(err)
	}
	for _, a := range []struct {
		name string
		r    image.Rectangle
	}{{"header-solid-ink", image.Rect(25, 45, 370, 68)}, {"entry-solid-ink", image.Rect(41, 428, 364, 610)}} {
		w := image.NewRGBA(src.Bounds())
		draw.Draw(w, w.Bounds(), src, image.Point{}, draw.Src)
		draw.Draw(w, a.r, &image.Uniform{color.RGBA{0, 0, 0, 255}}, image.Point{}, draw.Src)
		o, _ := os.Create(filepath.Join(os.Args[2], a.name+".png"))
		png.Encode(o, w)
		o.Close()
	}
	// tall-845.png: one page-colour row appended (390x845): fails only (iii′) dims
	b := src.Bounds()
	t := image.NewRGBA(image.Rect(0, 0, b.Dx(), b.Dy()+1))
	draw.Draw(t, t.Bounds(), &image.Uniform{src.At(5, 5)}, image.Point{}, draw.Src)
	draw.Draw(t, b, src, image.Point{}, draw.Src)
	o, _ := os.Create(filepath.Join(os.Args[2], "tall-845.png"))
	png.Encode(o, t)
	o.Close()
}
