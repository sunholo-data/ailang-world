def apply(rel, edits):
    p = os.path.join(root, rel)
    s = open(p, encoding="utf-8").read()
    for i, (old, new) in enumerate(edits):
        n = s.count(old)
        if n != 1:
            sys.exit(f"{rel}: edit {i + 1} matches {n} times, want 1: {old[:80]!r}")
        s = s.replace(old, new)
    return p, s

def line_edit(text, prefix, new):
    lines = text.split("\n")
    hits = [i for i, l in enumerate(lines) if l.startswith(prefix)]
    if len(hits) != 1:
        sys.exit(f"orchestration: line prefix {prefix!r} matches {len(hits)} lines, want 1")
    return (lines[hits[0]], new)
